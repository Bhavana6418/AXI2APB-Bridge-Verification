class base_test extends uvm_test;
 `uvm_component_utils(base_test)

//-------------declare handles----------//
environment envh;
env_config e_cfg;
axi_config axi_cfg[];
apb_config apb_cfg[];

int no_of_axi_agts=1;
int no_of_apb_agts=10;
int has_axi_agts=1;
int has_apb_agts=1;

int no_of_transactions;

bit [3:0] awlen=3;
bit [3:0] arlen=3;
//-------------------constructor new method--------------------//

function new(string name="base_test",uvm_component parent);
 super.new(name,parent);
endfunction

//-------------------function config--------------------//
function void config_bridge();
 if(has_axi_agts) 
   begin
     axi_cfg=new[no_of_axi_agts];
     foreach(axi_cfg[i])
      begin
       axi_cfg[i]=axi_config::type_id::create($sformatf("axi_cfg[%0d]",i));
       if(!uvm_config_db #(virtual axi_interface)::get(this,"",$sformatf("axi_if%0d",i),axi_cfg[i].axi_vif))
         `uvm_fatal("VIF CONFIG","cannot get()interface vif from uvm_config_db.have u set() it?")
          axi_cfg[i].is_active=UVM_ACTIVE;
          e_cfg.axi_cfg[i]=axi_cfg[i];
       end
    end

  if(has_apb_agts) 
   begin
     apb_cfg=new[no_of_apb_agts];
     foreach(apb_cfg[i])
      begin
       apb_cfg[i]=apb_config::type_id::create($sformatf("apb_cfg[%0d]",i));
       if(!uvm_config_db #(virtual apb_interface)::get(this,"",$sformatf("apb_if%0d",i),apb_cfg[i].vif))
         `uvm_fatal("VIF CONFIG","cannot get()interface vif from uvm_config_db.have u set() it?")
          apb_cfg[i].is_active=UVM_ACTIVE;
          e_cfg.apb_cfg[i]=apb_cfg[i];
       end
    end
     
    e_cfg.has_axi_agts=has_axi_agts;
    e_cfg.has_apb_agts=has_apb_agts;
    e_cfg.no_of_axi_agts=no_of_axi_agts;
    e_cfg.no_of_apb_agts=no_of_apb_agts;

endfunction

//--------------------function build phase-------------------//
function void build_phase(uvm_phase phase);
  e_cfg=env_config::type_id::create("e_cfg");

  if(has_axi_agts)
    e_cfg.axi_cfg=new[no_of_axi_agts];

  if(has_apb_agts)
    e_cfg.apb_cfg=new[no_of_apb_agts];

  config_bridge();

  uvm_config_db #(env_config)::set(this,"*","env_config",e_cfg);
  super.build_phase(phase);

  envh=environment::type_id::create("envh",this);
endfunction

function void end_of_elaboration_phase(uvm_phase phase);
 super.end_of_elaboration_phase(phase);
 uvm_top.print_topology;
endfunction

endclass
