		super.new(name,parent);
		axi_cg=new();
		axi_write_data_cg=new();
		axi_read_data_cg=new();
		apb_cg=new();
	endfunction

	function void build_phase(uvm_phase phase);
		super.build_phase(phase);

		if(!uvm_config_db #(env_config)::get(this," ","env_config",e_cfg))
			`uvm_fatal(get_type_name(),"cannot get the e_cfg from uvm_config_db,plz set_it")

		fifo_axi_awaddr_h=new[e_cfg.no_of_axi_agent];
		fifo_axi_araddr_h=new[e_cfg.no_of_axi_agent];
		fifo_apb_h=new[e_cfg.no_of_apb_agent];

		foreach(fifo_axi_awaddr_h[i])
			fifo_axi_awaddr_h[i]=new($sformatf("fifo_axi_awaddr_h[%0d]",i),this);


		foreach(fifo_axi_araddr_h[i])
			fifo_axi_araddr_h[i]=new($sformatf("fifo_axi_araddr_h[%0d]",i),this);

		foreach(fifo_apb_h[i])
			fifo_apb_h[i]=new($sformatf("fifo_apb_h[%0d]",i),this);

		foreach(e_cfg.apb_cfg[i])
			mvif=e_cfg.apb_cfg[i].mvif;

		foreach(e_cfg.apb_cfg[i])
			vif=e_cfg.apb_cfg[i].vif;
	endfunction

	task run_phase(uvm_phase phase);
		super.run_phase(phase);
		
		forever
		  
		  fork
			begin
				fifo_axi_awaddr_h[0].get(axi_xtnh);
				`uvm_info(get_type_name(),$sformatf("printing from subscriber axi write trnsaction \n %s",axi_xtnh.sprint()),UVM_NONE)

				axi_cg.sample();

				foreach(axi_xtnh.Wdata[i])
					axi_write_data_cg.sample(i);

			end

			begin
				fifo_axi_araddr_h[0].get(axi_xtnh);
				`uvm_info(get_type_name(),$sformatf("printing from subscriber axi read trnsaction \n %s",axi_xtnh.sprint()),UVM_NONE)


				foreach(axi_xtnh.Rdata[i])
					axi_read_data_cg.sample(i);

			end

			begin
				@(mvif.apb_mdrv_cb);
				wait(e_cfg.apb_cfg[0].mvif.Psel);
				@(mvif.apb_mdrv_cb);

					if(e_cfg.apb_cfg[0].vif.Pselx)
						begin
							fifo_apb_h[0].get(apb_xtnh);
							`uvm_info(get_type_name(),$sformatf("printing from subscriber apb_xtn \n %s",apb_xtnh.sprint()),UVM_NONE)
						end

 					if(e_cfg.apb_cfg[1].vif.Pselx)
						begin
							fifo_apb_h[1].get(apb_xtnh);
							`uvm_info(get_type_name(),$sformatf("printing from subscriber apb_xtn \n %s",apb_xtnh.sprint()),UVM_NONE)
						end

					if(e_cfg.apb_cfg[2].vif.Pselx)
						begin
							fifo_apb_h[2].get(apb_xtnh);
							`uvm_info(get_type_name(),$sformatf("printing from subscriber apb_xtn \n %s",apb_xtnh.sprint()),UVM_NONE)
						end

					if(e_cfg.apb_cfg[3].vif.Pselx)
						begin
							fifo_apb_h[3].get(apb_xtnh);
							`uvm_info(get_type_name(),$sformatf("printing from subscriber apb_xtn \n %s",apb_xtnh.sprint()),UVM_NONE)
						end

					if(e_cfg.apb_cfg[4].vif.Pselx)
						begin
							fifo_apb_h[4].get(apb_xtnh);
							`uvm_info(get_type_name(),$sformatf("printing from subscriber apb_xtn \n %s",apb_xtnh.sprint()),UVM_NONE)
						end

					if(e_cfg.apb_cfg[5].vif.Pselx)
						begin
							fifo_apb_h[5].get(apb_xtnh);
							`uvm_info(get_type_name(),$sformatf("printing from subscriber apb_xtn \n %s",apb_xtnh.sprint()),UVM_NONE)
						end

					if(e_cfg.apb_cfg[6].vif.Pselx)
						begin
							fifo_apb_h[6].get(apb_xtnh);
							`uvm_info(get_type_name(),$sformatf("printing from subscriber apb_xtn \n %s",apb_xtnh.sprint()),UVM_NONE)
						end

					if(e_cfg.apb_cfg[7].vif.Pselx)
						begin
							fifo_apb_h[7].get(apb_xtnh);
							`uvm_info(get_type_name(),$sformatf("printing from subscriber apb_xtn \n %s",apb_xtnh.sprint()),UVM_NONE)
						end

					if(e_cfg.apb_cfg[8].vif.Pselx)
						begin
							fifo_apb_h[8].get(apb_xtnh);
							`uvm_info(get_type_name(),$sformatf("printing from subscriber apb_xtn \n %s",apb_xtnh.sprint()),UVM_NONE)
						end

					if(e_cfg.apb_cfg[9].vif.Pselx)
						begin
							fifo_apb_h[9].get(apb_xtnh);
							`uvm_info(get_type_name(),$sformatf("printing from subscriber apb_xtn \n %s",apb_xtnh.sprint()),UVM_NONE)
						end

					apb_cg.sample;

			end

	  	  join_any
	
	endtask
