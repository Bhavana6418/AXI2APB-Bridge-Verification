class axi_agt_top extends uvm_env;
	`uvm_component_utils(axi_agt_top)
	
	env_config e_cfg;

	axi_agent axi_agt[];

function new(string name = "axi_agt_top",uvm_component parent);
	super.new(name,parent);
endfunction

function void build_phase(uvm_phase phase);
	super.build_phase(phase);
  	
	if(!uvm_config_db#(env_config)::get(this,"","env_config",e_cfg))
                         `uvm_fatal(get_type_name(),"configuration is not getting properly in environment ")

        if(e_cfg.has_axi_agts)
                begin
                          axi_agt = new[e_cfg.no_of_axi_agts];
                           foreach(axi_agt[i])
                                 begin
                                         axi_agt[i] = axi_agent::type_id::create($sformatf("axi_agt[%0d]",i),this);
                                         uvm_config_db#(axi_config)::set(this,$sformatf("axi_agt[%0d]*",i),"axi_config",e_cfg.axi_cfg[i]);
				end
			end

endfunction

endclass
