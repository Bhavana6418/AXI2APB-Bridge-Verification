class axi_sequence extends uvm_sequence #(axi_xtn);
	`uvm_object_utils(axi_sequence)

	bit[3:0] awlen=3;
	bit[3:0] arlen=3;

function new(string name = "axi_sequence");
	super.new(name);
endfunction

task body();
	if(!uvm_config_db#(bit[3:0])::get(null,get_full_name,"awlen",awlen))
		`uvm_fatal(get_type_name(),"cannot getting awlen from test")

	if(!uvm_config_db#(bit[3:0])::get(null,get_full_name,"arlen",arlen))
		`uvm_fatal(get_type_name(),"cannot getting arlen from test")
endtask

endclass

//axi_random sequence0
class axi_random_sequence extends axi_sequence;
	`uvm_object_utils(axi_random_sequence)

function new(string name = "axi_random_sequence");
	super.new(name);
endfunction

task body();
	super.body();
	repeat(1)
		begin
			req=axi_xtn::type_id::create("req");
			
			start_item(req);
			assert(req.randomize() with {Aresetn==1;Arlen==arlen;Awlen==awlen;Arvalid==1;});
			finish_item(req);
		end

endtask

endclass
